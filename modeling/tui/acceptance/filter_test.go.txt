package recordbrowser

// Copied unchanged into models whose fields are ID, Name, Status and
// Description, whose list label is "Records" and whose filter fields are Name
// and Description. TUI_ACCEPTANCE_VARIANT=hosted selects the hosted
// expectations (q and ctrl+c never quit); anything else is standalone.
import (
	"fmt"
	"os"
	"slices"
	"strings"
	"testing"

	tea "charm.land/bubbletea/v2"
	"github.com/charmbracelet/x/ansi"
)

func fhosted() bool { return os.Getenv("TUI_ACCEPTANCE_VARIANT") == "hosted" }

func frecs() []Record {
	return []Record{
		{ID: "a1", Name: "Alpha", Status: "active", Description: "First server"},
		{ID: "b2", Name: "Beta", Status: "active", Description: "Backup SERVER"},
		{ID: "g3", Name: "Gamma", Status: "active", Description: "Spare ÄBC"},
		{ID: "d4", Name: "Delta", Status: "active", Description: "xyz"},
	}
}

func fnew(t *testing.T, w, h int, r []Record) Model {
	t.Helper()
	m, err := New(r)
	if err != nil {
		t.Fatal(err)
	}
	next, _ := m.Update(tea.WindowSizeMsg{Width: w, Height: h})
	return next.(Model)
}

func fsend(m Model, msg tea.Msg) (Model, tea.Cmd) {
	next, cmd := m.Update(msg)
	return next.(Model), cmd
}

func fkey(m Model, k string) (Model, tea.Cmd) {
	v := tea.KeyPressMsg{}
	switch k {
	case "enter":
		v.Code = tea.KeyEnter
	case "esc":
		v.Code = tea.KeyEscape
	case "tab":
		v.Code = tea.KeyTab
	case "backspace":
		v.Code = tea.KeyBackspace
	case "down":
		v.Code = tea.KeyDown
	case "end":
		v.Code = tea.KeyEnd
	case "pgdown":
		v.Code = tea.KeyPgDown
	case "ctrl+c":
		v.Code, v.Mod = 'c', tea.ModCtrl
	default:
		v.Code, v.Text = []rune(k)[0], k
	}
	return fsend(m, v)
}

func fkeys(m Model, ks ...string) Model {
	for _, k := range ks {
		m, _ = fkey(m, k)
	}
	return m
}

// ftype delivers one key message per code point.
func ftype(m Model, s string) Model {
	for _, r := range s {
		m, _ = fkey(m, string(r))
	}
	return m
}

func fview(m Model) string { return ansi.Strip(m.View().Content) }

func TestFilterOpenTypeAndMatchRule(t *testing.T) {
	m := fnew(t, 80, 12, frecs())
	if m.FilterOpen() || m.CapturesKeys() || m.FilterText() != "" || m.Shown() != 4 {
		t.Fatal("initial filter state")
	}
	m, _ = fkey(m, "/")
	if !m.FilterOpen() || !m.CapturesKeys() || m.FilterText() != "" || m.Shown() != 4 {
		t.Fatal("open state")
	}
	m = ftype(m, "SERV")
	if m.FilterText() != "SERV" || m.Shown() != 2 || m.Selected() != "a1" {
		t.Fatalf("second field, case-insensitive: %q %d %q", m.FilterText(), m.Shown(), m.Selected())
	}
	v := fview(m)
	if !strings.Contains(v, "2/4 /SERV_") || strings.Contains(v, "Gamma") {
		t.Fatalf("label or rows:\n%s", v)
	}
	// ID and Status are not filter fields.
	m = fkeys(m, "esc")
	m = ftype(fkeys(m, "/"), "active")
	if m.Shown() != 0 {
		t.Fatalf("non-filter field matched: %d", m.Shown())
	}
	// Case folding is strings.ToLower: Ä folds to ä, there is no diacritic folding.
	m = fkeys(m, "esc")
	m = ftype(fkeys(m, "/"), "äb")
	if m.Shown() != 1 || m.Selected() != "g3" {
		t.Fatalf("Ä/ä: %d %q", m.Shown(), m.Selected())
	}
	m = fkeys(m, "esc")
	m = ftype(fkeys(m, "/"), "ab")
	if m.Shown() != 0 {
		t.Fatalf("diacritics folded: %d", m.Shown())
	}
	// No pattern syntax: . and * are literal characters.
	m = fkeys(m, "esc")
	m = ftype(fkeys(m, "/"), "a.")
	if m.Shown() != 0 {
		t.Fatal("pattern syntax")
	}
}

func TestFilterNoMatchesBackspaceKeepClear(t *testing.T) {
	m := fnew(t, 80, 12, frecs())
	m = ftype(fkeys(m, "/"), "zzz")
	if m.Shown() != 0 || m.Selected() != "" || !strings.Contains(fview(m), "No matches") {
		t.Fatalf("no matches: %d %q\n%s", m.Shown(), m.Selected(), fview(m))
	}
	// "z" matched Delta (xyz), so Delta was selected; it still matches the
	// empty filter and stays selected.
	m = fkeys(m, "backspace", "backspace", "backspace")
	if m.FilterText() != "" || m.Shown() != 4 || m.Selected() != "d4" {
		t.Fatalf("backspace: %q %d %q", m.FilterText(), m.Shown(), m.Selected())
	}
	m, _ = fkey(m, "backspace")
	if m.FilterText() != "" {
		t.Fatal("backspace on empty")
	}
	m = ftype(m, "ta")
	m, _ = fkey(m, "enter")
	if m.FilterOpen() || m.CapturesKeys() || m.FilterText() != "ta" || m.Shown() != 2 {
		t.Fatalf("enter keeps: %v %v %q %d", m.FilterOpen(), m.CapturesKeys(), m.FilterText(), m.Shown())
	}
	if !strings.Contains(fview(m), "[Records] 2/4 /ta") {
		t.Fatalf("kept label:\n%s", fview(m))
	}
	// esc in the list clears the kept filter; the selected record stays.
	m = fkeys(m, "down")
	sel := m.Selected()
	m, _ = fkey(m, "esc")
	if m.FilterText() != "" || m.Shown() != 4 || m.Selected() != sel {
		t.Fatalf("esc in list: %q %d %q (was %q)", m.FilterText(), m.Shown(), m.Selected(), sel)
	}
	// esc while typing clears the text and closes the input.
	m = ftype(fkeys(m, "/"), "be")
	m, _ = fkey(m, "esc")
	if m.FilterOpen() || m.FilterText() != "" || m.Shown() != 4 {
		t.Fatal("esc while typing")
	}
}

func TestFilterEscPrecedence(t *testing.T) {
	m := fnew(t, 80, 12, frecs())
	m = fkeys(ftype(fkeys(m, "/"), "server"), "enter")
	m = fkeys(m, "enter") // detail focus
	if m.Focus() != "detail" {
		t.Fatal("detail focus")
	}
	m, _ = fkey(m, "esc")
	if m.Focus() != "list" || m.FilterText() != "server" {
		t.Fatalf("first esc returns to the list: %s %q", m.Focus(), m.FilterText())
	}
	m, _ = fkey(m, "esc")
	if m.FilterText() != "" || m.Focus() != "list" {
		t.Fatalf("second esc clears: %q", m.FilterText())
	}
	// "/" from the detail pane opens the input on the list.
	m = fkeys(m, "enter", "/")
	if !m.FilterOpen() || m.Focus() != "list" {
		t.Fatal("/ from detail")
	}
	// esc with nothing selected clears a kept filter.
	m = fkeys(ftype(fkeys(fkeys(m, "esc"), "/"), "zzz"), "enter")
	if m.Selected() != "" || m.FilterText() != "zzz" {
		t.Fatal("setup")
	}
	m, _ = fkey(m, "esc")
	if m.FilterText() != "" || m.Selected() != "a1" {
		t.Fatalf("esc with no match: %q %q", m.FilterText(), m.Selected())
	}
}

func TestFilterInputCapturesEveryKey(t *testing.T) {
	m := fnew(t, 80, 12, frecs())
	m = fkeys(m, "/")
	for _, k := range []string{"q", "1", "j", "?", "k", "g", "G", " "} {
		var cmd tea.Cmd
		m, cmd = fkey(m, k)
		if cmd != nil {
			t.Fatalf("%q returned a command", k)
		}
	}
	if m.FilterText() != "q1j?kgG " || m.HelpOpen() || !m.FilterOpen() {
		t.Fatalf("text %q help %v", m.FilterText(), m.HelpOpen())
	}
	// Keys that type nothing do nothing.
	for _, k := range []string{"tab", "down", "end", "pgdown"} {
		var cmd tea.Cmd
		m, cmd = fkey(m, k)
		if cmd != nil || !m.FilterOpen() || m.FilterText() != "q1j?kgG " || m.Selected() != "" && m.Shown() > 0 {
			t.Fatalf("%q acted: %q open=%v", k, m.FilterText(), m.FilterOpen())
		}
	}
	m, _ = fkey(m, "esc")
	// With the input closed, q quits only a standalone browser.
	_, cmd := fkey(m, "q")
	if quit := cmd != nil && func() bool { _, ok := cmd().(tea.QuitMsg); return ok }(); quit == fhosted() {
		t.Fatalf("quit after closing: %v hosted %v", quit, fhosted())
	}
}

func TestFilterCtrlCQuitPolicy(t *testing.T) {
	base := fkeys(fnew(t, 80, 12, frecs()), "/")
	states := map[string]Model{"empty input": base, "typed input": ftype(base, "alp"), "no matches": ftype(base, "zzzz")}
	for _, status := range []Status{Loading, Failed, Ready} {
		m, err := base.Replace(Snapshot{Status: status})
		if err != nil {
			t.Fatal(err)
		}
		states[fmt.Sprintf("replacement %v", status)] = m
	}
	tiny, _ := base.Update(tea.WindowSizeMsg{Width: 1, Height: 1})
	states["tiny after opening"] = tiny.(Model)
	for name, before := range states {
		t.Run(name, func(t *testing.T) {
			if !before.FilterOpen() || !before.CapturesKeys() {
				t.Fatal("input must remain captured")
			}
			after, cmd := fkey(before, "ctrl+c")
			quit := false
			if cmd != nil {
				_, quit = cmd().(tea.QuitMsg)
			}
			if quit == fhosted() || (fhosted() && cmd != nil) {
				t.Fatalf("Ctrl+C quit=%v hosted=%v", quit, fhosted())
			}
			if after.browser.state != before.browser.state || fview(after) != fview(before) {
				t.Fatal("Ctrl+C changed input or browser state")
			}
		})
	}
	// Help remains captive; this policy only changes open filter input.
	help := fkeys(fnew(t, 80, 12, frecs()), "?")
	after, cmd := fkey(help, "ctrl+c")
	if cmd != nil || !after.HelpOpen() {
		t.Fatal("Ctrl+C escaped captive help")
	}
}

func TestFilterBoundsAndSanitising(t *testing.T) {
	m := fkeys(fnew(t, 80, 12, frecs()), "/")
	m = ftype(m, strings.Repeat("é", 70))
	if got := []rune(m.FilterText()); len(got) != 64 {
		t.Fatalf("bound: %d code points", len(got))
	}
	m = ftype(m, "x")
	if len([]rune(m.FilterText())) != 64 {
		t.Fatal("input beyond the bound was not ignored")
	}
	m, _ = fkey(m, "backspace")
	if got := m.FilterText(); len([]rune(got)) != 63 || strings.Contains(got, "�") {
		t.Fatalf("backspace removes one code point: %q", got)
	}
	// Multi-rune key text is bounded and control characters are dropped.
	m = fkeys(fnew(t, 80, 12, frecs()), "/")
	m, _ = fsend(m, tea.KeyPressMsg{Code: 'a', Text: "al\x1b[31m\x00\n\tpha"})
	if m.FilterText() != "al[31mpha" {
		t.Fatalf("sanitised: %q", m.FilterText())
	}
	m, _ = fsend(m, tea.KeyPressMsg{Code: 'x', Text: strings.Repeat("日本語", 40)})
	if len([]rune(m.FilterText())) != 64 {
		t.Fatalf("bounded: %d", len([]rune(m.FilterText())))
	}
	m, _ = fkey(m, "esc")
	// Paste is text too.
	m, _ = fkey(m, "/")
	m, _ = fsend(m, tea.PasteMsg{Content: "Be\nta\x00"})
	if m.FilterText() != "Beta" || m.Shown() != 1 {
		t.Fatalf("paste: %q %d", m.FilterText(), m.Shown())
	}
	// Paste is ignored while the input is closed.
	m = fkeys(m, "enter")
	m, _ = fsend(m, tea.PasteMsg{Content: "zzz"})
	if m.FilterText() != "Beta" {
		t.Fatal("paste while closed")
	}
}

// Valid replacement characters survive sanitizing; malformed bytes do not
// erase them or introduce extra replacement characters into the filter.
func TestFilterValidReplacementCharacter(t *testing.T) {
	records := []Record{{ID: "match", Name: "A\uFFFDB"}, {ID: "other", Name: "Other"}}
	for _, input := range []struct {
		name string
		send func(Model) Model
	}{
		{"typed", func(m Model) Model { return ftype(m, "A\uFFFDB") }},
		{"key text", func(m Model) Model {
			m, _ = fsend(m, tea.KeyPressMsg{Code: 'a', Text: "A\xff\uFFFD\xfe\n\t\x00\u2028\u2029B"})
			return m
		}},
		{"paste", func(m Model) Model {
			m, _ = fsend(m, tea.PasteMsg{Content: "A\xff\uFFFD\xfe\n\t\x00\u2028\u2029B"})
			return m
		}},
	} {
		t.Run(input.name, func(t *testing.T) {
			m := input.send(fkeys(fnew(t, 80, 12, records), "/"))
			if m.FilterText() != "A\uFFFDB" || m.Shown() != 1 || m.Selected() != "match" {
				t.Fatalf("text=%q shown=%d selected=%q", m.FilterText(), m.Shown(), m.Selected())
			}
			m = fkeys(m, "backspace", "backspace")
			if m.FilterText() != "A" {
				t.Fatalf("backspace did not remove one code point: %q", m.FilterText())
			}
		})
	}
	for _, paste := range []bool{false, true} {
		t.Run(fmt.Sprintf("cap paste=%v", paste), func(t *testing.T) {
			m := fkeys(fnew(t, 80, 12, records), "/")
			prefix := strings.Repeat("a", 63)
			text := prefix + "\xff\uFFFDx"
			if paste {
				m, _ = fsend(m, tea.PasteMsg{Content: text})
			} else {
				m, _ = fsend(m, tea.KeyPressMsg{Code: 'a', Text: text})
			}
			if m.FilterText() != prefix+"\uFFFD" {
				t.Fatalf("invalid bytes consumed room or valid replacement was dropped: %q", m.FilterText())
			}
			m = ftype(m, "\uFFFDz")
			if m.FilterText() != prefix+"\uFFFD" {
				t.Fatal("text beyond 64 code points was accepted")
			}
		})
	}
}

func TestFilterSelectionFollowsIdentityAndReplaceReapplies(t *testing.T) {
	m := fnew(t, 80, 12, frecs())
	m = fkeys(m, "down") // b2
	m = ftype(fkeys(m, "/"), "server")
	if m.Selected() != "b2" {
		t.Fatalf("selected record still matches: %q", m.Selected())
	}
	m = fkeys(ftype(m, "x"), "backspace")
	m = fkeys(m, "backspace", "backspace", "backspace", "backspace", "backspace", "backspace")
	m = ftype(m, "alp")
	if m.Selected() != "a1" {
		t.Fatalf("selection moves to the first match: %q", m.Selected())
	}
	m = fkeys(m, "enter")
	// Replace keeps and re-applies the filter.
	r := frecs()
	r = append(r, Record{ID: "e5", Name: "Epsilon-alp", Status: "active", Description: "new"})
	m, err := m.Replace(Snapshot{Status: Ready, Records: r})
	if err != nil || m.Shown() != 2 || m.FilterText() != "alp" || m.Selected() != "a1" {
		t.Fatalf("replace: %v %d %q %q", err, m.Shown(), m.FilterText(), m.Selected())
	}
	m, _ = m.Replace(Snapshot{Status: Loading})
	if m.FilterText() != "alp" || m.Selected() != "" {
		t.Fatal("loading keeps the filter text")
	}
	m, _ = m.Replace(Snapshot{Status: Failed, Message: "boom"})
	if m.FilterText() != "alp" {
		t.Fatal("failed keeps the filter text")
	}
	m, _ = m.Replace(Snapshot{Status: Ready, Records: []Record{{ID: "z", Name: "Zed", Status: "x", Description: "y"}}})
	if m.Shown() != 0 || m.Selected() != "" || !strings.Contains(fview(m), "No matches") {
		t.Fatalf("no record matches the kept filter: %d %q", m.Shown(), m.Selected())
	}
	m, _ = fkey(m, "esc")
	if m.FilterText() != "" || m.Selected() != "z" {
		t.Fatal("esc after replace")
	}
	// Clearing the filter keeps the selected identity.
	m = fnew(t, 80, 12, frecs())
	m = fkeys(ftype(fkeys(m, "/"), "delta"), "enter")
	m, _ = fkey(m, "esc")
	if m.Selected() != "d4" || m.Shown() != 4 {
		t.Fatalf("clear keeps identity: %q", m.Selected())
	}
}

func TestFilterNavigationUsesTheFilteredList(t *testing.T) {
	var r []Record
	for i := 0; i < 30; i++ {
		r = append(r, Record{ID: fmt.Sprintf("id-%02d", i), Name: fmt.Sprintf("Row-%02d", i), Status: "active", Description: "d"})
	}
	m := fnew(t, 80, 12, r)
	m = fkeys(ftype(fkeys(m, "/"), "row-2"), "enter")
	if m.Shown() != 10 || m.Selected() != "id-20" {
		t.Fatalf("shown %d selected %q", m.Shown(), m.Selected())
	}
	m, _ = fkey(m, "down")
	if m.Selected() != "id-21" {
		t.Fatalf("down: %q", m.Selected())
	}
	m, _ = fkey(m, "end")
	if m.Selected() != "id-29" {
		t.Fatalf("end: %q", m.Selected())
	}
	m, _ = fkey(m, "down")
	if m.Selected() != "id-29" {
		t.Fatalf("down at end: %q", m.Selected())
	}
	m, _ = fkey(m, "g")
	if m.Selected() != "id-20" {
		t.Fatalf("home: %q", m.Selected())
	}
	v := fview(m)
	if !strings.Contains(v, "10/30 /row-2") || strings.Contains(v, "Row-19") {
		t.Fatalf("filtered list:\n%s", v)
	}
	m, _ = fkey(m, "pgdown")
	if m.Selected() == "id-20" {
		t.Fatalf("pgdown did not move: %q", m.Selected())
	}
}

func TestFilterKeyHintsAndHelp(t *testing.T) {
	m := fnew(t, 80, 12, frecs())
	has := func(m Model, key, action string) bool {
		for _, h := range m.KeyHints() {
			if h.Key == key && h.Action == action {
				return true
			}
		}
		return false
	}
	if !has(m, "/", "filter") {
		t.Fatalf("hints %v", m.KeyHints())
	}
	m = fkeys(m, "/")
	h := m.KeyHints()
	want := []KeyHint{{"filter.keep", "enter", "keep"}, {"filter.cancel", "esc", "clear"}}
	if !fhosted() {
		want = append(want, KeyHint{"quit", "ctrl+c", "quit"})
	}
	if !slices.Equal(h, want) {
		t.Fatalf("input hints %v, want %v", h, want)
	}
	m = fkeys(ftype(m, "al"), "enter", "?")
	next, _ := m.Update(tea.WindowSizeMsg{Width: 80, Height: 40})
	m = next.(Model)
	help := fview(m)
	if !m.HelpOpen() || !strings.Contains(help, "filter the list") || !strings.Contains(help, "enter keeps, esc clears") {
		t.Fatalf("help:\n%s", help)
	}
	if strings.Contains(help, "ctrl+c quits while typing") == fhosted() {
		t.Fatalf("help quit policy hosted=%v:\n%s", fhosted(), help)
	}
	// Tiny: / does nothing.
	tiny := fnew(t, 10, 3, frecs())
	tiny, _ = fkey(tiny, "/")
	if tiny.FilterOpen() {
		t.Fatal("input opened in the tiny layout")
	}
	// Nothing to filter: / does nothing.
	empty := fnew(t, 80, 12, nil)
	empty, _ = fkey(empty, "/")
	if empty.FilterOpen() {
		t.Fatal("input opened without records")
	}
}

// Every hint the browser returns in any state has a stable ID from the
// generated vocabulary, no result repeats an ID, and across the states below
// the browser returns exactly the vocabulary (a hosted browser has no quit hint).
func TestFilterKeyHintIDs(t *testing.T) {
	seen := map[string]bool{}
	check := func(label string, m Model) {
		t.Helper()
		ids := map[string]bool{}
		for _, h := range m.KeyHints() {
			if h.ID == "" || h.Key == "" || h.Action == "" {
				t.Fatalf("%s: incomplete hint %+v", label, h)
			}
			if !slices.Contains(keyHintIDs, h.ID) {
				t.Fatalf("%s: id %q is not in the vocabulary %v", label, h.ID, keyHintIDs)
			}
			if ids[h.ID] {
				t.Fatalf("%s: id %q appears twice in %+v", label, h.ID, m.KeyHints())
			}
			ids[h.ID], seen[h.ID] = true, true
		}
	}
	m := fnew(t, 80, 12, frecs())
	check("list", m)
	d := fkeys(m, "enter")
	check("detail", d)
	check("help", fkeys(m, "?"))
	check("help from detail", fkeys(d, "?"))
	open := fkeys(m, "/")
	check("filter input", open)
	check("filter input with text", ftype(open, "al"))
	kept := fkeys(ftype(open, "al"), "enter")
	check("filter kept", kept)
	check("filter kept, detail", fkeys(kept, "enter"))
	check("filter kept, help", fkeys(kept, "?"))
	none := fkeys(ftype(fkeys(m, "/"), "zzz"), "enter")
	if none.Shown() != 0 {
		t.Fatalf("no-matches state shows %d records", none.Shown())
	}
	check("no matches", none)
	check("no matches, input", ftype(fkeys(m, "/"), "zzz"))
	for _, snap := range []Snapshot{{Status: Loading}, {Status: Failed, Message: "boom"}} {
		r, err := m.Replace(snap)
		if err != nil {
			t.Fatal(err)
		}
		check(fmt.Sprintf("status %v", snap.Status), r)
	}
	var got []string
	for id := range seen {
		got = append(got, id)
	}
	want := slices.Clone(keyHintIDs)
	if fhosted() {
		want = slices.DeleteFunc(want, func(id string) bool { return id == "quit" })
	}
	slices.Sort(got)
	slices.Sort(want)
	if !slices.Equal(got, want) {
		t.Fatalf("observed ids %v, want the documented vocabulary %v", got, want)
	}
}
